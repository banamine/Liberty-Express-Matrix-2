// ── Color palette ─────────────────────────────────────────────────────────────

  { bg: "bg-red-900/70",     border: "border-red-700",     text: "text-red-300",     dot: "bg-red-500"     },
  { bg: "bg-blue-900/70",    border: "border-blue-700",    text: "text-blue-300",    dot: "bg-blue-500"    },
  { bg: "bg-orange-900/70",  border: "border-orange-700",  text: "text-orange-300",  dot: "bg-orange-500"  },
  { bg: "bg-indigo-900/70",  border: "border-indigo-700",  text: "text-indigo-300",  dot: "bg-indigo-500"  },
  { bg: "bg-emerald-900/70", border: "border-emerald-700", text: "text-emerald-300", dot: "bg-emerald-500" },
  { bg: "bg-purple-900/70",  border: "border-purple-700",  text: "text-purple-300",  dot: "bg-purple-500"  },
  { bg: "bg-teal-900/70",    border: "border-teal-700",    text: "text-teal-300",    dot: "bg-teal-500"    },
  { bg: "bg-amber-900/70",   border: "border-amber-700",   text: "text-amber-300",   dot: "bg-amber-500"   },
  { bg: "bg-rose-900/70",    border: "border-rose-700",    text: "text-rose-300",    dot: "bg-rose-500"    },
  { bg: "bg-cyan-900/70",    border: "border-cyan-700",    text: "text-cyan-300",    dot: "bg-cyan-500"    },
];

function channelColor(groupTitle: string) {
  const g = groupTitle.toLowerCase();
  if (/fox/i.test(g)) return PALETTE[0];
  if (/cbs|kpix/i.test(g)) return PALETTE[1];
  if (/\brt\b|rt doc/i.test(g)) return PALETTE[2];
  if (/bbc/i.test(g)) return PALETTE[3];
  if (/ntd|newtang/i.test(g)) return PALETTE[4];
  if (/cnn/i.test(g)) return PALETTE[8];
  if (/nbc/i.test(g)) return PALETTE[6];
  const hash = Array.from(groupTitle).reduce((a, c) => a + c.charCodeAt(0), 0);
  return PALETTE[hash % PALETTE.length];
}

function channelAbbr(groupTitle: string): string {
  if (/fox/i.test(groupTitle)) return "FOX";
  if (/cbs|kpix/i.test(groupTitle)) return "CBS";
  if (/\brt\b|rt doc/i.test(groupTitle)) return "RT";
  if (/bbc/i.test(groupTitle)) return "BBC";
  if (/ntd|newtang/i.test(groupTitle)) return "NTD";
  if (/cnn/i.test(groupTitle)) return "CNN";
  if (/nbc/i.test(groupTitle)) return "NBC";
  return groupTitle.replace(/[^A-Z]/g, "").slice(0, 4) || groupTitle.slice(0, 3).toUpperCase();
}

// ── Date helpers ──────────────────────────────────────────────────────────────
function getSundayOfWeek(offset: number): Date {
  const now = new Date();
  const day = now.getDay();
  const sun = new Date(now);
  sun.setHours(0, 0, 0, 0);
  sun.setDate(now.getDate() - day + offset * 7);
  return sun;
}

function dateToYYYYMMDD(d: Date): string {
  return `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}${String(d.getDate()).padStart(2, "0")}`;
}

function fmtWeekRange(sun: Date): string {
  const sat = new Date(sun); sat.setDate(sun.getDate() + 6);
  const mo = sun.toLocaleDateString("en-US", { month: "short", day: "numeric" });
  const md = sat.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
  return `${mo} – ${md}`;
}

function fmtDayHeader(d: Date): string {
  return d.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
}

function fmtHour(h: number): string {
  if (h === 0) return "12a";
  if (h < 12) return `${h}a`;
  if (h === 12) return "12p";
  return `${h - 12}p`;
}

// ── Zoom presets ──────────────────────────────────────────────────────────────
// Dense:       1.4 px/min → 84px/hour  (fits more on screen, still legible)
// Comfortable: 2.0 px/min → 120px/hour (WCAG-friendly block heights)
const ZOOM_PRESETS = {
  dense:       { pxPerMin: 1.4, label: "Dense" },
  comfortable: { pxPerMin: 2.0, label: "Comfortable" },
} as const;
type ZoomMode = keyof typeof ZOOM_PRESETS;

// ── Episode parsing ───────────────────────────────────────────────────────────
interface PlacedBlock {
  ep: Episode;
  yyyymmdd: string;
  startH: number;
  durMin: number;
  lane: number;
  totalLanes: number;
}

function parseEpDateTime(ep: Episode): { yyyymmdd: string; startH: number } | null {
  const m = ep.id.match(/(\d{8})_(\d{6})/);
  if (!m) return null;
  const hh = parseInt(m[2].slice(0, 2), 10);
  const mm = parseInt(m[2].slice(2, 4), 10);
  return { yyyymmdd: m[1], startH: hh + mm / 60 };
}

// ── Lane assignment ───────────────────────────────────────────────────────────
function assignLanes(blocks: Array<{ ep: Episode; startH: number; durMin: number }>): PlacedBlock[] {
  const sorted = [...blocks].sort((a, b) => a.startH - b.startH);
  const laneEndH: number[] = [];

  const withLanes = sorted.map(block => {
    const endH = block.startH + block.durMin / 60;
    let lane = laneEndH.findIndex(end => end <= block.startH + 0.01);
    if (lane === -1) lane = laneEndH.length;
    laneEndH[lane] = endH;
    return { ...block, lane, totalLanes: 1 };
  });

  for (let i = 0; i < withLanes.length; i++) {
    const endH_i = withLanes[i].startH + withLanes[i].durMin / 60;
    let maxLane = withLanes[i].lane;
    for (let j = 0; j < withLanes.length; j++) {
      if (i === j) continue;
      const s = withLanes[j].startH, e = s + withLanes[j].durMin / 60;
      if (s < endH_i && e > withLanes[i].startH) maxLane = Math.max(maxLane, withLanes[j].lane);
    }
    withLanes[i].totalLanes = maxLane + 1;
  }

  return withLanes.map(b => ({
    ep: b.ep, yyyymmdd: (b as any).yyyymmdd ?? "",
    startH: b.startH, durMin: b.durMin,
    lane: b.lane, totalLanes: b.totalLanes,
  }));
}

// ── Static constants ──────────────────────────────────────────────────────────
const HOURS = Array.from({ length: 24 }, (_, i) => i);
const HEADER_H = 32;
const MAX_BLOCKS_PER_DAY = 40;
const DAY_LABELS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

const NAV_ITEMS = [
  { label: "Workbench",     icon: ArrowLeft, href: "/" },
  { label: "Archive.org",   icon: Globe,     href: "/" },
  { label: "Stream Finder", icon: Radio,     href: "/" },
  { label: "TV Player",     icon: Tv,        href: "/tvnews-player" },
  { label: "Live Player 2", icon: Tv,        href: "/live-player-2" },
  { label: "GitHub Sync",   icon: Github,    href: "/" },
  { label: "Holding Pen",   icon: Package,   href: "/" },
];

// ── Block component ───────────────────────────────────────────────────────────
function EpBlock({ b, dayTop, pxPerMin }: { b: PlacedBlock; dayTop: number; pxPerMin: number }) {
  const c = channelColor(b.ep.groupTitle ?? "");
  const abbr = channelAbbr(b.ep.groupTitle ?? "");
  const topPx = dayTop + b.startH * (60 * pxPerMin);
  const heightPx = Math.max(b.durMin * pxPerMin, 18);
  const widthPct = 100 / b.totalLanes;
  const leftPct = (b.lane / b.totalLanes) * 100;

  return (
    <div
      className={`absolute rounded border ${c.bg} ${c.border} px-1.5 overflow-hidden cursor-pointer hover:brightness-125 transition-[filter]`}
      style={{
        top: topPx,
        height: heightPx,
        left: `calc(2px + ${leftPct}%)`,
        width: `calc(${widthPct}% - 4px)`,
      }}
      title={`${b.ep.title} (${abbr}) — ${Math.round(b.durMin)}m`}
      data-testid={`cal-block-${b.ep.id}`}
    >
      <div className={`text-[10px] font-bold ${c.text} truncate leading-tight mt-0.5`}>{abbr}</div>
      {heightPx > 24 && (
        <div className="text-[10px] text-zinc-300 truncate leading-tight">{b.ep.title}</div>
      )}
      {heightPx > 40 && (
        <div className="text-[10px] text-zinc-500">{Math.round(b.durMin)}m</div>
      )}
    </div>
  );
}

// ── Main page ─────────────────────────────────────────────────────────────────
export default function CalendarGridPage() {
  const [, setLocation] = useLocation();
  const [weekOffset, setWeekOffset] = useState(0);
  const [hiddenChannels, setHiddenChannels] = useState<Set<string>>(new Set());
  const [zoom, setZoom] = useState<ZoomMode>("comfortable");

  const pxPerMin = ZOOM_PRESETS[zoom].pxPerMin;
  const HOUR_H   = 60 * pxPerMin;
  const TOTAL_H  = HOURS.length * HOUR_H;

  const { data: rawData, isLoading } = useQuery<{ episodes: Episode[] } | Episode[]>({
    queryKey: ["/api/episodes"],
    queryFn: async () => {
      const res = await fetch("/api/episodes");
      if (!res.ok) throw new Error("Failed to load episodes");
      return res.json();
    },
    staleTime: 60_000,
  });

  const episodes: Episode[] = useMemo(() => {
    if (!rawData) return [];
    return Array.isArray(rawData) ? rawData : (rawData as any).episodes ?? [];
  }, [rawData]);

  // ── Week dates ──────────────────────────────────────────────────────────────
  const weekStart = useMemo(() => getSundayOfWeek(weekOffset), [weekOffset]);
  const weekDays  = useMemo(() =>
    Array.from({ length: 7 }, (_, i) => {
      const d = new Date(weekStart); d.setDate(weekStart.getDate() + i);
      return { date: d, yyyymmdd: dateToYYYYMMDD(d), label: fmtDayHeader(d) };
    }), [weekStart]);

  const todayYYYYMMDD = useMemo(() => dateToYYYYMMDD(new Date()), []);

  // ── All unique channels ─────────────────────────────────────────────────────
  const allChannels = useMemo(() => {
    const seen = new Set<string>();
    for (const ep of episodes) if (ep.groupTitle) seen.add(ep.groupTitle);
    return Array.from(seen).sort();
  }, [episodes]);

  const toggleChannel = useCallback((ch: string) => {
    setHiddenChannels(prev => {
      const next = new Set(prev);
      if (next.has(ch)) next.delete(ch); else next.add(ch);
      return next;
    });
  }, []);

  // ── Block placement per day ─────────────────────────────────────────────────
  const blocksByDay = useMemo(() => {
    const byDay: Record<string, PlacedBlock[]> = {};
    for (const d of weekDays) byDay[d.yyyymmdd] = [];

    for (const ep of episodes) {
      if (ep.groupTitle && hiddenChannels.has(ep.groupTitle)) continue;
      const parsed = parseEpDateTime(ep);
      if (!parsed) continue;
      if (!byDay[parsed.yyyymmdd]) continue;
      const durMin = ep.duration ? ep.duration / 60 : 30;
      byDay[parsed.yyyymmdd].push({
        ep, yyyymmdd: parsed.yyyymmdd,
        startH: parsed.startH, durMin,
        lane: 0, totalLanes: 1,
      });
    }

    for (const yyyymmdd of Object.keys(byDay)) {
      const raw = byDay[yyyymmdd]
        .sort((a, b) => a.startH - b.startH)
        .slice(0, MAX_BLOCKS_PER_DAY);
      byDay[yyyymmdd] = assignLanes(raw);
    }
    return byDay;
  }, [episodes, weekDays, hiddenChannels]);

  const windowCount = useMemo(
    () => Object.values(blocksByDay).reduce((a: number, b: PlacedBlock[]) => a + b.length, 0),
    [blocksByDay]
  );

  return (
    <div className="flex h-screen bg-zinc-950 text-zinc-100 font-mono text-xs overflow-hidden">

      {/* ── Sidebar ────────────────────────────────────────────────────────── */}
      <aside className="w-48 shrink-0 bg-zinc-900 border-r border-zinc-800 flex flex-col">
        <div className="px-3 py-3 border-b border-zinc-800 flex items-center gap-2">
          <CalendarDays className="w-4 h-4 text-emerald-400 shrink-0" />
          <span className="text-emerald-400 font-bold text-sm tracking-wide">CALENDAR</span>
        </div>

        <nav className="px-2 pt-3">
          <p className="text-zinc-500 text-xs uppercase tracking-widest px-1 mb-1">Navigate</p>
          {NAV_ITEMS.map(({ label, icon: Icon, href }) => (
            <Link
              key={label}
              href={href}
              className="flex items-center gap-2 px-2 py-1.5 rounded text-zinc-400 hover:text-zinc-100 hover:bg-zinc-800 cursor-pointer text-xs"
              data-testid={`cal-nav-${label.toLowerCase().replace(/\s+/g, "-")}`}
            >
              <Icon className="w-3 h-3 shrink-0" />
              <span>{label}</span>
            </Link>
          ))}
        </nav>

        {/* Channel filters */}
        {allChannels.length > 0 && (
          <div className="mt-4 px-3 flex-1 overflow-y-auto">
            <p className="text-zinc-500 text-xs uppercase tracking-widest mb-2">Channels</p>
            {allChannels.map(ch => {
              const c = channelColor(ch);
              const hidden = hiddenChannels.has(ch);
              return (
                <div
                  key={ch}
                  onClick={() => toggleChannel(ch)}
                  data-testid={`cal-filter-${channelAbbr(ch)}`}
                  className={`flex items-center gap-1.5 px-1 py-1 rounded cursor-pointer hover:bg-zinc-800 ${hidden ? "opacity-40" : ""}`}
                >
                  <div className={`w-2 h-2 rounded-sm ${c.dot} shrink-0`} />
                  <span className={`text-xs font-medium ${c.text} truncate`}>{channelAbbr(ch)}</span>
                  <span className="text-zinc-500 text-[10px] truncate flex-1 min-w-0 ml-1">{ch.slice(0, 18)}</span>
                </div>
              );
            })}
          </div>
        )}

        <div className="px-3 py-3 border-t border-zinc-800 mt-auto">
          <div
            onClick={() => setLocation("/")}
            className="border border-dashed border-zinc-700 rounded p-2 text-center text-zinc-500 text-xs cursor-pointer hover:border-zinc-500 hover:text-zinc-400"
          >
            Drop m3u · json
          </div>
        </div>
      </aside>

      {/* ── Main ───────────────────────────────────────────────────────────── */}
      <div className="flex flex-col flex-1 min-w-0">

        {/* Toolbar */}
        <header className="h-10 border-b border-zinc-800 bg-zinc-900 flex items-center px-3 gap-2 shrink-0">
          <div className="flex gap-1.5 items-center flex-1">
            <span className="text-xs text-zinc-500 border border-zinc-700 rounded px-2 py-0.5">DEPLOY</span>
            <div className="flex gap-1 ml-2">
              {[1, 2, 3, 4, 5, 6].map(i => (
                <div key={i} className="w-5 h-5 rounded bg-zinc-800 border border-zinc-700" />
              ))}
            </div>
          </div>
          <div className="flex gap-2">
            <Link
              href="/player1"
              className="bg-emerald-500 text-black font-bold rounded px-3 py-1 text-xs hover:bg-emerald-400"
              data-testid="cal-news-loop"
            >
              PLAYER 1 (LINEAR)
            </Link>
            <Link
              href="/player2"
              className="bg-zinc-800 border border-zinc-600 rounded px-3 py-1 text-xs text-zinc-300 hover:bg-zinc-700"
              data-testid="cal-cinema-mode"
            >
              PLAYER 2 (AJ)
            </Link>
          </div>
        </header>

        {/* Week nav bar */}
        <div className="h-10 border-b border-zinc-800 bg-zinc-900/60 flex items-center px-3 gap-3 shrink-0">
          <button
            onClick={() => setWeekOffset(w => w - 1)}
            className="flex items-center gap-0.5 text-zinc-400 hover:text-zinc-100 text-xs px-1"
            data-testid="cal-prev-week"
          >
            <ChevronLeft className="w-3 h-3" /> Prev
          </button>

          <div className="flex gap-1">
            {["Week", "Day", "Schedule"].map((v, i) => (
              <div
                key={v}
                className={`px-2 py-0.5 rounded text-xs cursor-pointer ${
                  i === 0
                    ? "bg-emerald-900/40 text-emerald-400 border border-emerald-700"
                    : "bg-zinc-800 text-zinc-400 border border-zinc-700 hover:bg-zinc-700"
                }`}
              >
                {v}
              </div>
            ))}
          </div>

          <span className="text-zinc-300 text-xs font-semibold">{fmtWeekRange(weekStart)}</span>

          <button
            onClick={() => setWeekOffset(w => w + 1)}
            className="flex items-center gap-0.5 text-zinc-400 hover:text-zinc-100 text-xs px-1"
            data-testid="cal-next-week"
          >
            Next <ChevronRight className="w-3 h-3" />
          </button>

          <button
            onClick={() => setWeekOffset(0)}
            className="text-zinc-500 hover:text-zinc-300 text-xs border border-zinc-700 rounded px-1.5 py-0.5"
            data-testid="cal-today"
          >
            Today
          </button>

          {/* Zoom toggle */}
          <div className="flex gap-1 ml-2 items-center border border-zinc-700 rounded overflow-hidden">
            <button
              onClick={() => setZoom("dense")}
              className={`flex items-center gap-1 px-2 py-0.5 text-xs transition-colors ${
                zoom === "dense"
                  ? "bg-zinc-700 text-zinc-100"
                  : "bg-zinc-900 text-zinc-500 hover:text-zinc-300"
              }`}
              title="Dense view — more hours on screen"
              data-testid="cal-zoom-dense"
            >
              <AlignJustify className="w-3 h-3" />
              Dense
            </button>
            <button
              onClick={() => setZoom("comfortable")}
              className={`flex items-center gap-1 px-2 py-0.5 text-xs transition-colors ${
                zoom === "comfortable"
                  ? "bg-zinc-700 text-zinc-100"
                  : "bg-zinc-900 text-zinc-500 hover:text-zinc-300"
              }`}
              title="Comfortable view — readable block text"
              data-testid="cal-zoom-comfortable"
            >
              <LayoutList className="w-3 h-3" />
              Comfortable
            </button>
          </div>

          <span className="ml-auto text-zinc-500 text-xs">
            {isLoading ? "Loading…" : `${windowCount} episodes in window`}
          </span>
        </div>

        {/* Calendar grid */}
        <div className="flex-1 overflow-auto" data-testid="cal-scroll-area">
          <div className="flex" style={{ minWidth: 0 }}>

            {/* Time gutter */}
            <div className="w-12 shrink-0 relative bg-zinc-950" style={{ height: TOTAL_H + HEADER_H + 8 }}>
              <div className="sticky top-0 h-8 bg-zinc-950 z-20 border-b border-zinc-800" />
              {HOURS.map(h => (
                <div
                  key={h}
                  className="absolute right-2 text-[10px] text-zinc-500 select-none"
                  style={{ top: HEADER_H + h * HOUR_H - 6 }}
                >
                  {fmtHour(h)}
                </div>
              ))}
            </div>

            {/* Day columns */}
            {weekDays.map((day, di) => {
              const isToday = day.yyyymmdd === todayYYYYMMDD;
              const blocks = blocksByDay[day.yyyymmdd] ?? [];
              return (
                <div
                  key={day.yyyymmdd}
                  className={`flex-1 border-l border-zinc-800 min-w-0 relative ${isToday ? "bg-emerald-950/10" : ""}`}
                  style={{ height: TOTAL_H + HEADER_H + 8 }}
                  data-testid={`cal-day-${DAY_LABELS[di].toLowerCase()}`}
                >
                  {/* Day header */}
                  <div
                    className={`sticky top-0 z-10 flex items-center justify-center gap-1.5 border-b border-zinc-800 text-xs bg-zinc-950 ${
                      isToday ? "text-emerald-400" : "text-zinc-300"
                    }`}
                    style={{ height: HEADER_H }}
                  >
                    {day.label}
                    {blocks.length > 0 && (
                      <span className="text-zinc-500 text-[10px]">({blocks.length})</span>
                    )}
                  </div>

                  {/* Hour grid lines */}
                  {HOURS.map(h => (
                    <div
                      key={h}
                      className="absolute left-0 right-0 border-t border-zinc-800/40"
                      style={{ top: HEADER_H + h * HOUR_H }}
                    />
                  ))}

                  {/* Half-hour guide lines (subtle) */}
                  {HOURS.map(h => (
                    <div
                      key={`h${h}`}
                      className="absolute left-0 right-0 border-t border-zinc-800/20"
                      style={{ top: HEADER_H + h * HOUR_H + HOUR_H / 2 }}
                    />
                  ))}

                  {/* Episode blocks */}
                  {blocks.map((b, bi) => (
                    <EpBlock key={bi} b={b} dayTop={HEADER_H} pxPerMin={pxPerMin} />
                  ))}
                </div>
              );
            })}
          </div>
        </div>

        {/* Status bar */}
        <footer className="h-7 border-t border-zinc-800 bg-zinc-900 flex items-center px-3 gap-4 shrink-0">
          <span className="text-emerald-400 text-[10px]">Ready</span>
          <span className="text-zinc-500 text-[10px]">Total: {isLoading ? "…" : episodes.length}</span>
          <span className="text-zinc-500 text-[10px]">
            Viewing week of {weekStart.toLocaleDateString("en-US", { month: "short", day: "numeric" })}
          </span>
          {hiddenChannels.size > 0 && (
            <span className="text-zinc-500 text-[10px]">
              {hiddenChannels.size} channel{hiddenChannels.size > 1 ? "s" : ""} hidden
            </span>
          )}
          <span className="ml-auto text-zinc-600 text-[10px]">
            {ZOOM_PRESETS[zoom].label} · {pxPerMin} px/min
          </span>
        </footer>
      </div>
    </div>
  );
}
